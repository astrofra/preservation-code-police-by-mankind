cmake_minimum_required(VERSION 3.16)
project(CodePolice LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CODEPOLICE_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
find_package(Threads REQUIRED)
find_package(SDL2 CONFIG REQUIRED)
add_library(codepolice-core src/engine.cpp src/assets.cpp)
target_include_directories(codepolice-core PUBLIC src)
target_include_directories(codepolice-core SYSTEM PRIVATE vendor/stb)
target_link_libraries(codepolice-core PUBLIC Threads::Threads)
if(MSVC)
    target_compile_options(codepolice-core PRIVATE /W4 /fp:strict)
else()
    target_compile_options(codepolice-core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    # Preserved decompiler locals/parameters and explicit AST parentheses.
    set_source_files_properties(src/engine.cpp PROPERTIES COMPILE_OPTIONS "-Wno-unused-parameter;-Wno-unused-variable;-Wno-parentheses-equality")
endif()
if(CODEPOLICE_SANITIZERS AND NOT MSVC)
    add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(codepolice-core PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    add_link_options(-fsanitize=address,undefined)
endif()
add_executable(verify-native tools/verify.cpp)
target_link_libraries(verify-native PRIVATE codepolice-core)
add_executable(code-police src/main.cpp)
target_link_libraries(code-police PRIVATE codepolice-core SDL2::SDL2)
if(TARGET SDL2::SDL2main)
    target_link_libraries(code-police PRIVATE SDL2::SDL2main)
endif()
add_custom_target(codepolice-assets ALL COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:code-police>/assets")
add_dependencies(codepolice-assets code-police)
